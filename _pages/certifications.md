<script>
(function(){
  var modal=document.getElementById('cert-modal');
  if(!modal)return;
  document.body.appendChild(modal); // lift out of #main so it covers the fixed masthead
  var img=modal.querySelector('.cert-modal__img');
  var cap=modal.querySelector('.cert-modal__caption');
  var link=modal.querySelector('.cert-modal__link');
  function open(full,caption){
    img.setAttribute('src',full);img.setAttribute('alt',caption);
    cap.textContent=caption;link.setAttribute('href',full);
    modal.hidden=false;document.body.classList.add('cert-modal-open');
    modal.querySelector('.cert-modal__close').focus();
  }
  function close(){modal.hidden=true;document.body.classList.remove('cert-modal-open');img.setAttribute('src','');}
  document.querySelectorAll('.cert-item').forEach(function(b){
    b.addEventListener('click',function(){open(b.getAttribute('data-cert-full'),b.getAttribute('data-cert-caption'));});
  });
  modal.querySelectorAll('[data-cert-close]').forEach(function(el){el.addEventListener('click',close);});
  document.addEventListener('keydown',function(e){if(e.key==='Escape'&&!modal.hidden){close();}});
})();
</script>
