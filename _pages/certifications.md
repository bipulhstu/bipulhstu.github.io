<script>
(function(){
  var modal=document.getElementById('cert-modal');
  if(!modal)return;
  document.body.appendChild(modal); /* lift out of #main so it covers the fixed masthead */
  var img=modal.querySelector('.cert-modal__img');
  var cap=modal.querySelector('.cert-modal__caption');
  var link=modal.querySelector('.cert-modal__link');
  var verify=modal.querySelector('.cert-modal__verify');
  function open(full,caption,verifyUrl){
    img.setAttribute('src',full);img.setAttribute('alt',caption);
    cap.textContent=caption;link.setAttribute('href',full);
    if(verifyUrl){verify.setAttribute('href',verifyUrl);verify.hidden=false;}
    else{verify.hidden=true;verify.removeAttribute('href');}
    modal.hidden=false;document.body.classList.add('cert-modal-open');
    modal.querySelector('.cert-modal__close').focus();
  }
  function close(){modal.hidden=true;document.body.classList.remove('cert-modal-open');img.setAttribute('src','');}
  document.querySelectorAll('.cert-item').forEach(function(b){
    b.addEventListener('click',function(){open(b.getAttribute('data-cert-full'),b.getAttribute('data-cert-caption'),b.getAttribute('data-cert-verify'));});
  });
  modal.querySelectorAll('[data-cert-close]').forEach(function(el){el.addEventListener('click',close);});
  document.addEventListener('keydown',function(e){if(e.key==='Escape'&&!modal.hidden){close();}});
})();
</script>

<script>
(function(){
  var btns=document.querySelectorAll('.cert-viewbtn');
  var grids=document.querySelectorAll('.cert-grid');
  if(!btns.length||!grids.length)return;
  function setView(view){
    var isList=view==='list';
    grids.forEach(function(g){g.classList.toggle('cert-grid--list',isList);});
    btns.forEach(function(b){
      var active=b.getAttribute('data-view')===view;
      b.classList.toggle('is-active',active);
      b.setAttribute('aria-pressed',active?'true':'false');
    });
    try{localStorage.setItem('certView',view);}catch(e){}
  }
  btns.forEach(function(b){b.addEventListener('click',function(){setView(b.getAttribute('data-view'));});});
  var saved=null;
  try{saved=localStorage.getItem('certView');}catch(e){}
  if(saved==='list'){setView('list');}
})();
</script>
